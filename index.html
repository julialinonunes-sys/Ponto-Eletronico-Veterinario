// ATUALIZAÇÃO E CORREÇÃO DO CÁLCULO EXATO DE HORAS (hh:mm:ss) function updateCoordinationDashboard() { const selectedMonth = document.getElementById('coordFilterMonth').value; const coordSelectVal = document.getElementById('coordFilterVet').value; let targetVet = currentUser.accessLevel === "Coordenação" ? coordSelectVal : currentUser.name; const filteredLogs = logs.filter(log => { const logMonth = log.timestamp ? log.timestamp.slice(0, 7) : ""; const matchMonth = selectedMonth ? logMonth === selectedMonth : true; const matchVet = targetVet !== "ALL" ? log.vetName === targetVet : true; return matchMonth && matchVet; }); // Agrupa os registros por Profissional e por Data const grouped = {}; filteredLogs.forEach(log => { const key = `${log.vetName}_${log.formattedDate}`; if (!grouped[key]) { grouped[key] = { vetName: log.vetName, vetRole: log.vetRole, date: log.formattedDate, logs: [], justificativas: [] }; } grouped[key].logs.push(log); if (log.reasonCategory && log.reasonCategory !== "Sem alteração (No horário)") { let desc = log.reasonCategory; if(log.notes) desc += `: ${log.notes}`; grouped[key].justificativas.push(`[${log.type}] ${desc}`); } }); const rowsArray = Object.values(grouped); const tbody = document.getElementById('coordLogsTable'); if (!tbody) return; let grandTotalSecondsRegular = 0; let grandTotalSecondsPlantao = 0; let grandTotalOcorrencias = 0; if (rowsArray.length === 0) { tbody.innerHTML = `Nenhum ponto registrado para o período selecionado.`; } else { tbody.innerHTML = rowsArray.map(row => { // Ordena os registros do dia pelo timestamp cronológico row.logs.sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp)); let entrada = null, saidaAlmoco = null, retornoAlmoco = null, saida = null; let plantaoEntrada = null, plantaoSaida = null; let secReg = 0; let secPlantao = 0; row.logs.forEach(l => { const t = new Date(l.timestamp); if (l.type === "Entrada") entrada = t; if (l.type === "Saída Almoço") saidaAlmoco = t; if (l.type === "Retorno Almoço") retornoAlmoco = t; if (l.type === "Saída") saida = t; if (l.type === "Entrada Plantão") plantaoEntrada = t; if (l.type === "Saída Plantão") plantaoSaida = t; }); // 1. Período da Manhã (Entrada até Saída Almoço) if (entrada && saidaAlmoco && saidaAlmoco > entrada) { secReg += (saidaAlmoco - entrada) / 1000; } // 2. Período da Tarde (Retorno Almoço até Saída) if (retornoAlmoco && saida && saida > retornoAlmoco) { secReg += (saida - retornoAlmoco) / 1000; } // 3. Turno Direto (Entrada até Saída sem pausa de almoço) if (entrada && saida && !saidaAlmoco && !retornoAlmoco && saida > entrada) { secReg += (saida - entrada) / 1000; } // 4. Período de Plantão (Entrada Plantão até Saída Plantão) if (plantaoEntrada && plantaoSaida && plantaoSaida > plantaoEntrada) { secPlantao += (plantaoSaida - plantaoEntrada) / 1000; } grandTotalSecondsRegular += secReg; grandTotalSecondsPlantao += secPlantao; grandTotalOcorrencias += row.justificativas.length; const secDiaTotal = secReg + secPlantao; const fmtEntrada = entrada ? entrada.toLocaleTimeString('pt-BR') : '-'; const fmtSaidaAlmoco = saidaAlmoco ? saidaAlmoco.toLocaleTimeString('pt-BR') : '-'; const fmtRetornoAlmoco = retornoAlmoco ? retornoAlmoco.toLocaleTimeString('pt-BR') : '-'; const fmtSaida = saida ? saida.toLocaleTimeString('pt-BR') : '-'; let fmtPlantao = '-'; if (plantaoEntrada || plantaoSaida) { const pIn = plantaoEntrada ? plantaoEntrada.toLocaleTimeString('pt-BR') : '--:--'; const pOut = plantaoSaida ? plantaoSaida.toLocaleTimeString('pt-BR') : '--:--'; fmtPlantao = `${pIn} às ${pOut}`; } const fmtJustificativas = row.justificativas.length > 0 ? `
${row.justificativas.join('
')}
` : `-`; return ` ${row.vetName}
${row.vetRole} ${row.date} ${fmtEntrada} ${fmtSaidaAlmoco} ${fmtRetornoAlmoco} ${fmtSaida} ${fmtPlantao} ${formatSecondsToHHMMSS(secDiaTotal)} ${fmtJustificativas} `; }).join(''); } document.getElementById('cardTotalRegular').textContent = formatSecondsToHHMMSS(grandTotalSecondsRegular); document.getElementById('cardTotalPlantao').textContent = formatSecondsToHHMMSS(grandTotalSecondsPlantao); document.getElementById('cardTotalOcorrencias').textContent = `${grandTotalOcorrencias} ocorrências`; document.getElementById('cardTotalGeral').textContent = formatSecondsToHHMMSS(grandTotalSecondsRegular + grandTotalSecondsPlantao); }
