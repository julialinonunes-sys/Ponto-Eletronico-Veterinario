Carregar perfil do utilizador logado na coleção 'vets' async function loadUserProfile(user) { try { const vetDocRef = window.doc(window.firebaseDb, 'vets', user.uid); const vetDocSnap = await window.getDoc(vetDocRef); if (vetDocSnap.exists()) { currentUser = { uid: user.uid, ...vetDocSnap.data() }; } else { // Fallback de busca por e-mail caso o ID do documento seja diferente do UID const q = window.query( window.collection(window.firebaseDb, 'vets'), window.where('email', '==', user.email) ); const querySnap = await window.getDocs(q); if (!querySnap.empty) { const docData = querySnap.docs[0]; currentUser = { uid: user.uid, docId: docData.id, ...docData.data() }; } else { // Perfil padrão caso não encontre o documento currentUser = { uid: user.uid, email: user.email, name: user.email, accessLevel: 'Colaborador', role: 'Colaborador' }; } } updateUIForUser(); } catch (error) { console.error("Erro ao carregar perfil da coleção vets:", error); } } // 2. Salvar ou atualizar cadastro de veterinário no Painel da Coordenação async function saveVetForm(event) { event.preventDefault(); const vetId = document.getElementById('vet-id').value; const vetData = { name: document.getElementById('vet-name').value, userName: document.getElementById('vet-username').value, email: document.getElementById('vet-email').value, role: document.getElementById('vet-role').value, accessLevel: document.getElementById('vet-access').value, status: document.getElementById('vet-status').value, updatedAt: new Date().toISOString() }; try { if (vetId) { // Atualizar existente na coleção 'vets' await window.updateDoc(window.doc(window.firebaseDb, 'vets', vetId), vetData); alert('Cadastro atualizado com sucesso!'); } else { // Criar novo registro na coleção 'vets' vetData.createdAt = new Date().toISOString(); await window.addDoc(window.collection(window.firebaseDb, 'vets'), vetData); alert('Veterinário cadastrado com sucesso!'); } closeVetModal(); loadVetsList(); // Recarrega a tabela de colaboradores } catch (error) { console.error("Erro ao salvar na coleção vets:", error); alert("Falha ao salvar cadastro: " + error.message); } } // 3. Carregar lista de colaboradores no Painel da Coordenação async function loadVetsList() { const vetsTableBody = document.getElementById('vets-table-body'); if (!vetsTableBody) return; vetsTableBody.innerHTML = 'A carregar...'; try { const querySnap = await window.getDocs(window.collection(window.firebaseDb, 'vets')); vetsTableBody.innerHTML = ''; querySnap.forEach((docSnap) => { const data = docSnap.data(); const id = docSnap.id; const row = document.createElement('tr'); row.innerHTML = ` ${data.name || '-'} ${data.email || '-'} ${data.role || '-'} ${data.accessLevel || 'Colaborador'} ${data.status || 'Ativo'} `; vetsTableBody.appendChild(row); }); } catch (error) { console.error("Erro ao carregar lista da coleção vets:", error); vetsTableBody.innerHTML = 'Erro ao carregar lista.'; } } // 4. Excluir cadastro da coleção 'vets' async function deleteVet(docId) { if (confirm("Tem certeza que deseja excluir este cadastro?")) { try { await window.deleteDoc(window.doc(window.firebaseDb, 'vets', docId)); alert("Cadastro removido com sucesso!"); loadVetsList(); } catch (error) { console.error("Erro ao excluir da coleção vets:", error); alert("Erro ao excluir cadastro."); } } }
